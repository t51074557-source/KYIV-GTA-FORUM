<script>

function showUsers() {

    const users =
        JSON.parse(
            localStorage.getItem("KYIV_GTA_FORUM_USERS")
            || "[]"
        );

    const box =
        document.getElementById("admin-content");

    box.innerHTML =
        "<h2>Користувачі</h2>" +

        users.map(u => `

            <div class="post">

                <div class="post-body">

                    <b>${u.username}</b>

                    <br>

                    ${u.email}

                    <br>

                    Ранг: ${u.rank}

                </div>

            </div>

        `).join("");
}


function clearTopics() {

    if (!confirm("Видалити всі теми?")) {
        return;
    }

    localStorage.removeItem(
        "KYIV_GTA_FORUM_TOPICS"
    );

    alert("Теми видалено.");

    location.reload();
}

</script>
